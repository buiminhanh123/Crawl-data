function toVarName(str) {
  var map = {
    'à':'a','á':'a','ả':'a','ã':'a','ạ':'a','ă':'a','ắ':'a','ặ':'a','ằ':'a','ẳ':'a','ẵ':'a',
    'â':'a','ấ':'a','ậ':'a','ầ':'a','ẩ':'a','ẫ':'a','đ':'d',
    'è':'e','é':'e','ẻ':'e','ẽ':'e','ẹ':'e','ê':'e','ế':'e','ệ':'e','ề':'e','ể':'e','ễ':'e',
    'ì':'i','í':'i','ỉ':'i','ĩ':'i','ị':'i',
    'ò':'o','ó':'o','ỏ':'o','õ':'o','ọ':'o','ô':'o','ố':'o','ộ':'o','ồ':'o','ổ':'o','ỗ':'o',
    'ơ':'o','ớ':'o','ợ':'o','ờ':'o','ở':'o','ỡ':'o',
    'ù':'u','ú':'u','ủ':'u','ũ':'u','ụ':'u','ư':'u','ứ':'u','ự':'u','ừ':'u','ử':'u','ữ':'u',
    'ỳ':'y','ý':'y','ỷ':'y','ỹ':'y','ỵ':'y'
  };
  return str.toLowerCase()
    .split('').map(function(c){ return map[c]||c; }).join('')
    .replace(/[^a-z0-9]+/g,'-')
    .replace(/^-+|-+$/g,'');
}

function addVar(name, col) {
  var tbody = document.getElementById('varBody');
  var tr = document.createElement('tr');
  tr.className = 'var-row';
  tr.innerHTML =
    '<td><input type="text" placeholder="VD: Thuong hieu" value="'+(name||'')+'" oninput="updateToken(this)"></td>' +
    '<td><input type="text" placeholder="A" value="'+(col||'')+'" style="text-align:center"></td>' +
    '<td><span class="var-token" onclick="copyToken(this)">{'+toVarName(name||'')+'}</span></td>' +
    '<td><button class="btn-del" onclick="delVar(this)">✕</button></td>';
  tbody.appendChild(tr);
}

function updateToken(input) {
  var row = input.closest('tr');
  var token = row.querySelector('.var-token');
  var slug = toVarName(input.value);
  token.textContent = slug ? '{'+slug+'}' : '{...}';
  token.className = 'var-token';
}

function delVar(btn) { btn.closest('tr').remove(); }

function copyToken(el) {
  var text = el.textContent;
  var ta = document.getElementById('prompt');
  if (document.activeElement === ta) {
    var s = ta.selectionStart, e = ta.selectionEnd;
    ta.value = ta.value.substring(0,s) + text + ta.value.substring(e);
    ta.selectionStart = ta.selectionEnd = s + text.length;
    ta.focus();
  } else {
    var tmp = document.createElement('textarea');
    tmp.value = text; document.body.appendChild(tmp);
    tmp.select(); document.execCommand('copy'); document.body.removeChild(tmp);
  }
  el.textContent = '✓ copied!'; el.className = 'var-token copied';
  setTimeout(function(){ el.textContent = text; el.className = 'var-token'; }, 1200);
}

function getVariables() {
  var vars = [];
  document.querySelectorAll('#varBody .var-row').forEach(function(row) {
    var inputs = row.querySelectorAll('input');
    var name = inputs[0].value.trim(), col = inputs[1].value.trim();
    if (name && col) vars.push({ varName: toVarName(name), col: col });
  });
  return vars;
}

// ── SLOTS & PROGRESS ──
function renderSlots(n) {
  var w = document.getElementById('batchSlots');
  w.innerHTML = '';
  for (var i = 0; i < n; i++) {
    var s = document.createElement('div');
    s.className='slot'; s.id='slot-'+i; w.appendChild(s);
  }
}

function updateSlots() {
  // Reset tất cả về idle trước
  var slots = document.querySelectorAll('.slot');
  slots.forEach(function(s){ s.className='slot'; });

  // Map theo index thứ tự trong activeSlots
  var entries = Object.keys(state.activeSlots);
  for (var i = 0; i < entries.length && i < slots.length; i++) {
    var status = state.activeSlots[entries[i]];
    slots[i].className = 'slot ' + status;
  }
}

function updateProgress() {
  var pct = state.total>0 ? Math.round((state.done+state.errors)/state.total*100) : 0;
  document.getElementById('progressBar').style.width = pct+'%';
  document.getElementById('doneCount').textContent   = state.done;
  document.getElementById('errorCount').textContent  = state.errors;
  document.getElementById('runningCount').textContent = Object.keys(state.activeSlots)
    .filter(function(k){return state.activeSlots[k]==='running';}).length;
  document.getElementById('totalCount').textContent  = state.total;
}

function setStatus(s) {
  var b = document.getElementById('statusBadge');

}

function pad(n){return n<10?'0'+n:n;}
function fmtTime(ms){
  if(ms<1000) return ms+'ms';
  if(ms<60000) return (ms/1000).toFixed(1)+'s';
  return Math.floor(ms/60000)+'p '+Math.round((ms%60000)/1000)+'s';
}

// ── MAIN PROCESS ──
function startProcess() {
  var prompt         = document.getElementById('prompt').value.trim();
  var destCol        = document.getElementById('destCol').value.trim();
  var startRow       = parseInt(document.getElementById('startRow').value)||3;
  var endRowRaw      = document.getElementById('endRow').value.trim();
  var endRow         = endRowRaw ? parseInt(endRowRaw) : 0;
  var batchSize      = parseInt(document.getElementById('batchSize').value)||5;
  var sheetName      = document.getElementById('sheetName').value;
  var extraBlacklist = document.getElementById('extraBlacklist').value.trim();
  var maxFields      = parseInt(document.getElementById('maxFields').value)||0;
  var maxChars       = parseInt(document.getElementById('maxChars').value)||0;
  var moc            = parseInt(document.getElementById('maxOutputChars').value)||0;
  var variables      = getVariables();

  if (!prompt)            { alert('Vui lòng nhập Prompt!'); return; }
  if (!destCol)           { alert('Vui lòng nhập Cột đích!'); return; }
  if (variables.length===0){ alert('Vui lòng thêm ít nhất 1 biến!'); return; }

  state.running=true; state.stop=false; state.rows=[];
  state.nextIndex=0; state.activeSlots={};
  state.batchSize=batchSize; state.done=0; state.errors=0; state.total=0;
  state.startTime=new Date().getTime(); state.rowTimes=[];

  document.getElementById('btnStart').disabled=true;
  document.getElementById('btnStop').disabled=false;
  setStatus('running'); clearLog(); renderSlots(batchSize);

  addLog('🚀 Bắt đầu | Sheet: '+sheetName+' → Cột: '+destCol,'system');
  addLog('Biến: '+variables.map(function(v){return '{'+v.varName+'}='+v.col;}).join(', '),'system');
  addLog('Batch:'+batchSize+' | Field≤'+(maxFields||'∞')+' | Ký tự≤'+(maxChars||'∞')+' | Output≤'+(moc||'∞'),'system');

  google.script.run
    .withSuccessHandler(function(rows) {
      if (!rows||rows.length===0){ addLog('⚠️ Không có dữ liệu!','error'); finishProcess(); return; }
      state.rows  = rows.filter(function(r){return r.data&&r.data.trim();});
      state.total = state.rows.length;
      addLog('📊 Tổng hàng: '+state.total,'info');
      updateProgress();
      fillSlots(prompt,variables,destCol,sheetName,extraBlacklist,maxFields,maxChars,moc);
    })
    .withFailureHandler(function(err){ addLog('❌ Lỗi: '+err,'error'); finishProcess(); })
    .getSheetData(sheetName, variables[0].col, startRow, endRow);
}

function fillSlots(prompt,variables,destCol,sheetName,eb,mf,mc,moc) {
  if (state.stop){ checkDone(prompt,variables,destCol,sheetName,eb,mf,mc,moc); return; }
  var running = Object.keys(state.activeSlots).filter(function(k){return state.activeSlots[k]==='running';}).length;
  while (running < state.batchSize && state.nextIndex < state.rows.length) {
    if (state.stop) break;
    var r = state.rows[state.nextIndex++]; running++;
    state.activeSlots[r.row]='running';
    updateSlots(); updateProgress();
    addLog('⏳ Hàng '+r.row+'...','running');
    runRow(r.row,prompt,variables,destCol,sheetName,eb,mf,mc,moc);
  }
  checkDone(prompt,variables,destCol,sheetName,eb,mf,mc,moc);
}

function runRow(row,prompt,variables,destCol,sheetName,eb,mf,mc,moc) {
  var rowStart = new Date().getTime();
  google.script.run
    .withSuccessHandler(function(res) {
      var elapsed = new Date().getTime() - rowStart;
      if (res.success) {
        state.done++; state.activeSlots[res.row]='done';
        // Xóa khỏi activeSlots sau 1.5s để slot về idle
        if (!res.skipped) {
          google.script.run.clearErrorCell(sheetName, res.row, destCol);
        }
        setTimeout(function(){ 
          delete state.activeSlots[res.row]; 
          updateSlots(); 
        }, 2000);
        if (res.skipped) {
          addLog('⏭ Hàng '+res.row+': bỏ qua (đã có dữ liệu)','info');
        } else {
          state.rowTimes.push(elapsed);
          addLog('✅ Hàng '+res.row+' ('+fmtTime(elapsed)+')','success');
        }
      } else {
        state.errors++; state.activeSlots[res.row]='error';
        setTimeout(function(){ 
          delete state.activeSlots[res.row]; 
          updateSlots(); 
        }, 2000);
        addLog('❌ Hàng '+res.row+': '+res.error,'error');
      }
      updateSlots(); updateProgress();
      setTimeout(function(){ fillSlots(prompt,variables,destCol,sheetName,eb,mf,mc,moc); },300);
    })
    .withFailureHandler(function(err) {
      state.errors++; state.activeSlots[row]='error';
      addLog('❌ Hàng '+row+': '+err,'error');
      updateSlots(); updateProgress();
      setTimeout(function(){ fillSlots(prompt,variables,destCol,sheetName,eb,mf,mc,moc); },300);
    })
    .processRow({
      sheetName:sheetName, row:row, variables:variables,
      destCol:destCol, prompt:prompt, extraBlacklist:eb,
      maxFields:mf, maxChars:mc, maxOutputChars:moc,
      skipExisting:document.getElementById('skipExisting').checked
    });
}

function checkDone(prompt,variables,destCol,sheetName,eb,mf,mc,moc) {
  var running = Object.keys(state.activeSlots).filter(function(k){return state.activeSlots[k]==='running';}).length;
  if (state.nextIndex>=state.rows.length && running===0) finishProcess();
}

function finishProcess() {
  state.running=false; state.stop=false;
  document.getElementById('btnStart').disabled=false;
  document.getElementById('btnStop').disabled=true;
  document.getElementById('btnRetry').disabled=false;
  setStatus(state.stop?'stopped':'done');
  var totalTime = new Date().getTime() - state.startTime;
  var times = state.rowTimes;
  if (times.length>0) {
    var avg=Math.round(times.reduce(function(a,b){return a+b;},0)/times.length);
    addLog('🎉 Xong! Tổng:'+state.total+' ✓'+state.done+' ✗'+state.errors,'success');
    addLog('⏱ '+fmtTime(totalTime)+' | TB:'+fmtTime(avg)+'/hàng | Nhanh:'+fmtTime(Math.min.apply(null,times))+' | Chậm:'+fmtTime(Math.max.apply(null,times)),'info');
  } else {
    addLog('🎉 Xong! Tổng:'+state.total+' ✓'+state.done+' ✗'+state.errors,'success');
  }
  if (state.isRetry) {
  google.script.run.clearErrorRows(document.getElementById('sheetName').value);
  state.isRetry = false;
}
  saveHistoryEntry();
  state.errors>0 ? playErrorSound() : playDoneSound();
  setTimeout(function(){ state.activeSlots={}; updateSlots(); },2000);
}

function stopProcess() {
  state.stop=true;
  addLog('⏸ Đang dừng...','info');
  document.getElementById('btnStart').disabled=false;
  document.getElementById('btnStop').disabled=true;
}

// ── RETRY ERRORS ──
function retryErrors() {
  var sheetName = document.getElementById('sheetName').value;
  var moc       = parseInt(document.getElementById('maxOutputChars').value)||0;
  var destCol   = document.getElementById('destCol').value.trim();
  var startRow  = parseInt(document.getElementById('startRow').value)||3;
  var endRow    = parseInt(document.getElementById('endRow').value)||0;
  if (!destCol) { alert('Vui lòng nhập Cột đích!'); return; }

  addLog('🔍 Tìm hàng lỗi trong cache...','system');

  // Thử cache trước
  google.script.run
    .withSuccessHandler(function(cachedRows) {
      if (cachedRows && cachedRows.length > 0) {
        addLog('✅ Cache: '+cachedRows.length+' hàng lỗi','info');
        runRetry(cachedRows);
      } else {
        // Không có cache → scan sheet theo batch 200
        addLog('⚠️ Cache rỗng → scan sheet theo batch 200 hàng...','info');
        scanBatchForErrors(sheetName, destCol, startRow, endRow, startRow, []);
      }
    })
    .withFailureHandler(function(err){ addLog('❌ '+err,'error'); })
    .getErrorRowsFromTag(sheetName);
}

function scanBatchForErrors(sheetName, destCol, startRow, endRow, batchStart, accumulated) {
  addLog('🔍 Đang scan hàng '+batchStart+'...','system');
  google.script.run
    .withSuccessHandler(function(result) {
      var all = accumulated.concat(result.errorRows);
      if (result.hasMore) {
        // Còn batch tiếp theo → scan tiếp
        setTimeout(function() {
          scanBatchForErrors(sheetName, destCol, startRow, endRow, result.nextBatch, all);
        }, 500);
      } else {
        // Hết rồi → chạy retry
        addLog('📊 Tìm thấy '+all.length+' hàng lỗi/trống','info');
        if (all.length === 0) {
          addLog('✅ Không có hàng lỗi nào!','success'); return;
        }
        runRetry(all);
      }
    })
    .withFailureHandler(function(err){ addLog('❌ Lỗi scan: '+err,'error'); })
    .getErrorRowsBatch(sheetName, destCol, startRow, endRow, batchStart);
}

function runRetry(errorRows) {
  var sheetName = document.getElementById('sheetName').value;
  var destCol   = document.getElementById('destCol').value.trim();
  state.isRetry = true;
  addLog('🔁 Bắt đầu retry '+errorRows.length+' hàng...','info');

  state.running=true; state.stop=false;
  state.rows=errorRows; state.nextIndex=0; state.activeSlots={};
  state.batchSize=parseInt(document.getElementById('batchSize').value)||5;
  state.done=0; state.errors=0; state.total=errorRows.length;
  state.startTime=new Date().getTime(); state.rowTimes=[];

  document.getElementById('btnStart').disabled=true;
  document.getElementById('btnStop').disabled=false;
  document.getElementById('btnRetry').disabled=true;
  setStatus('running'); renderSlots(state.batchSize); updateProgress();

  var prompt         = document.getElementById('prompt').value.trim();
  var variables      = getVariables();
  var extraBlacklist = document.getElementById('extraBlacklist').value.trim();
  var maxFields      = parseInt(document.getElementById('maxFields').value)||0;
  var maxChars       = parseInt(document.getElementById('maxChars').value)||0;
  var moc            = parseInt(document.getElementById('maxOutputChars').value)||0;

  fillSlots(prompt,variables,destCol,sheetName,extraBlacklist,maxFields,maxChars,moc);
}

      var osc=ctx.createOscillator(), gain=ctx.createGain();
      osc.connect(gain); gain.connect(ctx.destination);
      osc.frequency.value=freq; osc.type='sine';
      gain.gain.setValueAtTime(0.7, ctx.currentTime+i*.15);
      gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime+i*.15+.3);
      osc.start(ctx.currentTime+i*.15); osc.stop(ctx.currentTime+i*.15+.3);
    });
  } catch(e){}
}
